;== LEGEND ==
<dreg> = A | B | C | D
<ireg> = X | Y | L | H
Z = Y:X
SP = H:L
<r16> = Z | SP
;== ====== ==

;========   SYSTEM   ========
HALT ; Trigger HALT
NOP  ; Do Nothing

TRAP #n2 ; Trigger Nth TRAP vector

FSET #n3 ; Set bit in Status Register
FCLR #n3 ; Clear bit in Status Register

BIT <dreg>, #imm8 ; <dreg> & #imm8, update SR from results, no writeback
BIT <dreg>, abs16 ; <dreg> & mem[abs16], update SR from results, no writeback
BIT <dreg>, [Z]   ; <dreg> & mem[Z], update SR from results, no writeback

;========  TRANSFER  ========
; These do not use extension byte
MOVI <dreg>, #imm8 ; dreg <- mem[PC]
MOVI <ireg>, #imm8 ; ireg <- mem[PC]
MOVI <r16>, #imm16 ; r16 <- mem[PC+1]:mem[PC]

MOV <dreg>, SR  ; dreg <- SR
MOV <ireg>, VBR ; ireg <- VBR

MOV SR, <dreg>  ; SR <- dreg
MOV VBR, <ireg> ; VBR <- ireg

; Those do use extension byte
EXG <dreg>, <ireg>   ; dreg <-> ireg
EXG <dregx>, <dregy> ; dregx <-> dregy
EXG <iregx>, <iregy> ; iregx <-> iregy

MOV <dregx>, <dregy> ; dregx <- dregy
MOV <iregx>, <iregy> ; iregx <- iregy
MOV <dregx>, <iregy> ; dregx <- iregy
MOV <iregx>, <dregy> ; iregx <- dregy

MOV <dreg>, abs16 ; dreg <- mem[abs16]
MOV <ireg>, abs16 ; ireg <- mem[abs16]
MOV <r16>, abs16  ; r16 <- mem[abs16+1]:mem[abs16]

MOV <dreg>, [imm8:<ireg>] ; dreg <- mem[imm8:ireg]
MOV <dreg>, [<r16>]       ; dreg <- mem[r16]
MOV <dreg>, [<r16>+]      ; dreg <- mem[r16], increment r16
MOV <dreg>, [-<r16>]      ; decrement r16, dreg <- mem[r16]

MOV abs16, <dreg> ; mem[abs16] <- dreg
MOV abs16, <ireg> ; mem[abs16] <- ireg
MOV abs16, <r16>  ; mem[abs16] <- r16[7:0], mem[abs16+1] <- r16[15:8]

MOV [imm8:<ireg>], <dreg> ; mem[imm8:ireg] <- dreg
MOV [<r16>], <dreg>       ; mem[r16] <- dreg
MOV [<r16>+], <dreg>      ; mem[r16] <- dreg, increment r16
MOV [-<r16>], <dreg>      ; decrement r16, mem[r16] <- dreg

;======== ARITHMETIC ========
; These do not use extension byte
ADCI <dreg>, #imm8 ; dreg <- dreg + mem[PC] + C
ADDI <dreg>, #imm8 ; dreg <- dreg + mem[PC]
SBCI <dreg>, #imm8 ; dreg <- dreg - mem[PC] - (~C)
SUBI <dreg>, #imm8 ; dreg <- dreg - mem[PC]
ANDI <dreg>, #imm8 ; dreg <- dreg & mem[PC]
ORI  <dreg>, #imm8 ; dreg <- dreg | mem[PC]
XORI <dreg>, #imm8 ; dreg <- dreg ^ mem[PC]
CMPI <dreg>, #imm8 ; dreg - mem[PC], update SR from results, no writeback

; These do use extension byte
ADC <dregx>, <dregy> ; dregx <- dregx + dregy + C
ADD <dregx>, <dregy> ; dregx <- dregx + dregy
SBC <dregx>, <dregy> ; dregx <- dregx - dregy - (~C)
SUB <dregx>, <dregy> ; dregx <- dregx - dregy
AND <dregx>, <dregy> ; dregx <- dregx & dregy
OR  <dregx>, <dregy> ; dregx <- dregx | dregy
XOR <dregx>, <dregy> ; dregx <- dregx ^ dregy
CMP <dregx>, <dregy> ; dregx - dregy, update SR from results, no writeback

ADC <dreg>, abs16 ; dreg <- dreg + mem[abs16] + C
ADD <dreg>, abs16 ; dreg <- dreg + mem[abs16]
SBC <dreg>, abs16 ; dreg <- dreg - mem[abs16] - (~C)
SUB <dreg>, abs16 ; dreg <- dreg - mem[abs16]
AND <dreg>, abs16 ; dreg <- dreg & mem[abs16]
OR  <dreg>, abs16 ; dreg <- dreg | mem[abs16]
XOR <dreg>, abs16 ; dreg <- dreg ^ mem[abs16]
CMP <dreg>, abs16 ; dreg - mem[abs16], update SR from results, no writeback

ADC <dreg>, [imm8:<ireg>] ; dreg <- dreg + mem[imm8:ireg] + C
ADD <dreg>, [imm8:<ireg>] ; dreg <- dreg + mem[imm8:ireg]
SBC <dreg>, [imm8:<ireg>] ; dreg <- dreg - mem[imm8:ireg] - (~C)
SUB <dreg>, [imm8:<ireg>] ; dreg <- dreg - mem[imm8:ireg]
CMP <dreg>, [imm8:<ireg>] ; dreg - mem[imm8:ireg], update SR from results, no writeback

ADC <dreg>, [Z] ; dreg <- dreg + mem[Z] + C
ADD <dreg>, [Z] ; dreg <- dreg + mem[Z]
SBC <dreg>, [Z] ; dreg <- dreg - mem[Z] - (~C)
SUB <dreg>, [Z] ; dreg <- dreg - mem[Z]
CMP <dreg>, [Z] ; dreg - mem[Z], update SR from results, no writeback

ADC <dreg>, [Z+] ; dreg <- dreg + mem[Z] + C, increment Z
ADD <dreg>, [Z+] ; dreg <- dreg + mem[Z], increment Z
SBC <dreg>, [Z+] ; dreg <- dreg - mem[Z] - (~C), increment Z
SUB <dreg>, [Z+] ; dreg <- dreg - mem[Z], increment Z
CMP <dreg>, [Z+] ; dreg - mem[Z], increment Z, update SR from results, no writeback

ADC <dreg>, [-Z] ; decrement Z, dreg <- dreg + mem[Z] + C
ADD <dreg>, [-Z] ; decrement Z, dreg <- dreg + mem[Z]
SBC <dreg>, [-Z] ; decrement Z, dreg <- dreg - mem[Z] - (~C)
SUB <dreg>, [-Z] ; decrement Z, dreg <- dreg - mem[Z]
CMP <dreg>, [-Z] ; decrement Z, dreg - mem[Z], update SR from results, no writeback

MULU <dregx>, <dregy> ; y = x * y; dregx <- y[7:0]; dregy <- y[15:8]
DIVU <dregx>, <dregy> ; q = x/y; r = x%y; dregx <- q; dregy <- r
MULS <dregx>, <dregy> ; signed multiply; dregx <- y[7:0]; dregy <- y[15:8]
DIVS <dregx>, <dregy> ; signed divide; r = x%y; dregx <- q; dregy <- r

; These do not use extension byte
INC <dreg> ; dreg <- dreg + 1
DEC <dreg> ; dreg <- dreg - 1
ROL <dreg> ; dreg <- dreg <] 1
ROR <dreg> ; dreg <- dreg [> 1
ASR <dreg> ; dreg <- dreg >> 1, sign-extended
LSL <dreg> ; dreg <- dreg << 1
LSR <dreg> ; dreg <- dreg >> 1
NOT <dreg> ; dreg <- ~dreg
CLR <dreg> ; dreg <- $00
NEG <dreg> ; dreg <- $00 - dreg

INC <ireg> ; ireg <- ireg + 1
DEC <ireg> ; ireg <- ireg - 1
CLR <ireg> ; ireg <- $00

;========  CONTROL   ========
RTI ; PULL SR, PULL PC
RTS ; PULL PC

JMP imm16       ; PC <- mem[PC+1]:mem[PC]
JMP imm8:<ireg> ; PC <- mem[imm8:ireg+1]:mem[imm8:ireg]
JMP Z           ; PC <- Z

JSR imm16       ; PUSH PC, PC <- mem[PC+1]:mem[PC]
JSR imm8:<ireg> ; PUSH PC, PC <- mem[imm8:ireg+1]:mem[imm8:ireg]
JSR Z           ; PUSH PC, PC <- Z

BRA rel8 ; PC <- PC + rel8
BSR rel8 ; PUSH PC, PC <- PC + rel8

BPL rel8 ; IF N=0 THEN PC <- PC + rel8
BVC rel8 ; IF V=0 THEN PC <- PC + rel8
BCC rel8 ; IF C=0 THEN PC <- PC + rel8
BNE rel8 ; IF Z=0 THEN PC <- PC + rel8
BMI rel8 ; IF N=1 THEN PC <- PC + rel8
BVS rel8 ; IF V=1 THEN PC <- PC + rel8
BCS rel8 ; IF C=1 THEN PC <- PC + rel8
BEQ rel8 ; IF Z=1 THEN PC <- PC + rel8
