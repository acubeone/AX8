;== LEGEND ==
<dreg> = A | B | C | D
<ireg> = X | Y | L | H
Z = Y:X
SP = H:L
<r16> = Z | SP
<r8> = SR | VBR
;== ====== ==

;== ADDRMODE ==
DataRegister   | <dreg>            ; Single do not use Extesnion
IndexRegister  | <ireg>            ; Single do not use Extension
RegisterPair   | <r16>             ; Single do not use Extension
SystemRegister | <r8>              ; Doesn't use Extension
Implied        | ... or #n         ; Doesn't use Extension
Immediate      | #imm8/#imm16/rel8 ; Doesn't use Extension
Absolute       | abs16
PagedAddress   | [imm8:<ireg>]     ; Single not use Extension
Indirect       | [<r16>]
PostIncrement  | [<r16>+]
PreDecrement   | [-<r16>]

;========   SYSTEM   ========
HALT ; Trigger HALT
NOP  ; Do Nothing

TRAP #n2 ; Trigger Nth TRAP vector

BCLR <dreg>, #imm8 ; dreg <- dreg & ~mask, mask = 1 << imm8[2:0]
BSET <dreg>, #imm8 ; dreg <- dreg | mask, mask = 1 << imm8[2:0]

BCLR abs16, <dreg> ; mem[abs16] <- mem[abs16] & ~mask, mask = 1 << dreg[2:0]
BSET abs16, <dreg> ; mem[abs16] <- mem[abs16] | mask, mask = 1 << dreg[2:0]

BIT <dreg>, #imm8   ; 0 <- <dreg> & #imm8, update SR from results
BIT <dreg>, abs16   ; 0 <- <dreg> & mem[abs16], update SR from results
BIT <dreg>, [<r16>] ; 0 <- <dreg> & mem[r16], update SR from results

;========  TRANSFER  ========
; These do not use extension byte
MOVI <dreg>, #imm8 ; dreg <- mem[PC]
MOVI <ireg>, #imm8 ; ireg <- mem[PC]
MOVI <r16>, #imm16 ; r16 <- mem[PC+1]:mem[PC]

MOV <dreg>, SR  ; dreg <- SR
MOV <ireg>, VBR ; ireg <- VBR

MOV SR, <dreg>  ; SR <- dreg
MOV VBR, <ireg> ; VBR <- ireg

; Those do use extension byte
EXG <dreg>, <ireg>   ; dreg <-> ireg
EXG <dregx>, <dregy> ; dregx <-> dregy
EXG <iregx>, <iregy> ; iregx <-> iregy

MOV <dregx>, <dregy> ; dregx <- dregy
MOV <iregx>, <iregy> ; iregx <- iregy
MOV <dregx>, <iregy> ; dregx <- iregy
MOV <iregx>, <dregy> ; iregx <- dregy

MOV <dreg>, abs16 ; dreg <- mem[abs16]
MOV <ireg>, abs16 ; ireg <- mem[abs16]
MOV <r16>, abs16  ; r16 <- mem[abs16+1]:mem[abs16]

MOV <dreg>, [imm8:<ireg>] ; dreg <- mem[imm8:ireg]
MOV <dreg>, [<r16>]       ; dreg <- mem[r16]
MOV <dreg>, [<r16>+]      ; dreg <- mem[r16], increment r16
MOV <dreg>, [-<r16>]      ; decrement r16, dreg <- mem[r16]

MOV abs16, <dreg> ; mem[abs16] <- dreg
MOV abs16, <ireg> ; mem[abs16] <- ireg
MOV abs16, <r16>  ; mem[abs16] <- r16[7:0], mem[abs16+1] <- r16[15:8]

MOV [imm8:<ireg>], <dreg> ; mem[imm8:ireg] <- dreg
MOV [<r16>], <dreg>       ; mem[r16] <- dreg
MOV [<r16>+], <dreg>      ; mem[r16] <- dreg, increment r16
MOV [-<r16>], <dreg>      ; decrement r16, mem[r16] <- dreg

;======== ARITHMETIC ========
; These do not use extension byte
ADCI <dreg>, #imm8 ; dreg <- dreg + mem[PC] + C
SBCI <dreg>, #imm8 ; dreg <- dreg - mem[PC] - (~C)
ANDI <dreg>, #imm8 ; dreg <- dreg & mem[PC]
ORI  <dreg>, #imm8 ; dreg <- dreg | mem[PC]
XORI <dreg>, #imm8 ; dreg <- dreg ^ mem[PC]
ADDI <dreg>, #imm8 ; dreg <- dreg + mem[PC]
SUBI <dreg>, #imm8 ; dreg <- dreg - mem[PC]
CMPI <dreg>, #imm8 ; 0 <- dreg - mem[PC], update SR from results

; These do use extension byte
ADC <dregx>, <dregy> ; dregx <- dregx + dregy + C
SBC <dregx>, <dregy> ; dregx <- dregx - dregy - (~C)
AND <dregx>, <dregy> ; dregx <- dregx & dregy
OR  <dregx>, <dregy> ; dregx <- dregx | dregy
XOR <dregx>, <dregy> ; dregx <- dregx ^ dregy
ADD <dregx>, <dregy> ; dregx <- dregx + dregy
SUB <dregx>, <dregy> ; dregx <- dregx - dregy
CMP <dregx>, <dregy> ; 0 <- dregx - dregy, update SR from results

ADC <dreg>, abs16 ; dreg <- dreg + mem[abs16] + C
SBC <dreg>, abs16 ; dreg <- dreg - mem[abs16] - (~C)
AND <dreg>, abs16 ; dreg <- dreg & mem[abs16]
OR  <dreg>, abs16 ; dreg <- dreg | mem[abs16]
XOR <dreg>, abs16 ; dreg <- dreg ^ mem[abs16]
ADD <dreg>, abs16 ; dreg <- dreg + mem[abs16]
SUB <dreg>, abs16 ; dreg <- dreg - mem[abs16]
CMP <dreg>, abs16 ; 0 <- dreg - mem[abs16], update SR from results

ADC <dreg>, [imm8:<ireg>] ; dreg <- dreg + mem[imm8:ireg] + C
SBC <dreg>, [imm8:<ireg>] ; dreg <- dreg - mem[imm8:ireg] - (~C)
AND <dreg>, [imm8:<ireg>] ; dreg <- dreg & mem[imm8:ireg]
OR  <dreg>, [imm8:<ireg>] ; dreg <- dreg | mem[imm8:ireg]
XOR <dreg>, [imm8:<ireg>] ; dreg <- dreg ^ mem[imm8:ireg]
ADD <dreg>, [imm8:<ireg>] ; dreg <- dreg + mem[imm8:ireg]
SUB <dreg>, [imm8:<ireg>] ; dreg <- dreg - mem[imm8:ireg]
CMP <dreg>, [imm8:<ireg>] ; 0 <- dreg - mem[imm8:ireg], update SR from results

ADC <dreg>, [<r16>] ; dreg <- dreg + mem[r16] + C
SBC <dreg>, [<r16>] ; dreg <- dreg - mem[r16] - (~C)
AND <dreg>, [<r16>] ; dreg <- dreg & mem[r16]
OR  <dreg>, [<r16>] ; dreg <- dreg | mem[r16]
XOR <dreg>, [<r16>] ; dreg <- dreg ^ mem[r16]
ADD <dreg>, [<r16>] ; dreg <- dreg + mem[r16]
SUB <dreg>, [<r16>] ; dreg <- dreg - mem[r16]
CMP <dreg>, [<r16>] ; 0 <- dreg - mem[r16], update SR from results

ADC <dreg>, [<r16>+] ; dreg <- dreg + mem[r16] + C, increment r16
SBC <dreg>, [<r16>+] ; dreg <- dreg - mem[r16] - (~C), increment r16
AND <dreg>, [<r16>+] ; dreg <- dreg & mem[r16], increment r16
OR  <dreg>, [<r16>+] ; dreg <- dreg | mem[r16], increment r16
XOR <dreg>, [<r16>+] ; dreg <- dreg ^ mem[r16], increment r16
ADD <dreg>, [<r16>+] ; dreg <- dreg + mem[r16], increment r16
SUB <dreg>, [<r16>+] ; dreg <- dreg - mem[r16], increment r16
CMP <dreg>, [<r16>+] ; 0 <- dreg - mem[r16], increment r16, update SR from results

ADC <dreg>, [-<r16>] ; decrement r16, dreg <- dreg + mem[r16] + C
SBC <dreg>, [-<r16>] ; decrement r16, dreg <- dreg - mem[r16] - (~C)
AND <dreg>, [-<r16>] ; decrement r16, dreg <- dreg & mem[r16]
OR  <dreg>, [-<r16>] ; decrement r16, dreg <- dreg | mem[r16]
XOR <dreg>, [-<r16>] ; decrement r16, dreg <- dreg ^ mem[r16]
ADD <dreg>, [-<r16>] ; decrement r16, dreg <- dreg + mem[r16]
SUB <dreg>, [-<r16>] ; decrement r16, dreg <- dreg - mem[r16]
CMP <dreg>, [-<r16>] ; decrement r16, 0 <- dreg - mem[r16], update SR from results

MULU <dregx>, <dregy> ; y = x * y; dregx <- y[7:0]; dregy <- y[15:8]
DIVU <dregx>, <dregy> ; q = x/y; r = x%y; dregx <- q; dregy <- r
MULS <dregx>, <dregy> ; signed multiply; dregx <- y[7:0]; dregy <- y[15:8]
DIVS <dregx>, <dregy> ; signed divide; r = x%y; dregx <- q; dregy <- r

; These do not use extension byte
INC <dreg> ; dreg <- dreg + 1
DEC <dreg> ; dreg <- dreg - 1
ROL <dreg> ; dreg <- dreg <] 1
ROR <dreg> ; dreg <- dreg [> 1
NOT <dreg> ; dreg <- ~dreg
ASR <dreg> ; dreg <- dreg >> 1, sign-extended
LSL <dreg> ; dreg <- dreg << 1
LSR <dreg> ; dreg <- dreg >> 1
CLR <dreg> ; dreg <- $00
NEG <dreg> ; dreg <- $00 - dreg

INC <ireg> ; ireg <- ireg + 1
DEC <ireg> ; ireg <- ireg - 1
CLR <ireg> ; ireg <- $00

;========  CONTROL   ========
RTS ; PULL PC
RTI ; PULL SR, PULL PC

JSR abs16   ; PUSH PC, PC <- mem[PC+1]:mem[PC]
JSR [<r16>] ; PUSH PC, PC <- mem[r16+1]:mem[r16]

JMP abs16   ; PC <- mem[PC+1]:mem[PC]
JMP [<r16>] ; PC <- mem[r16+1]:mem[r16]

BSR rel8 ; PUSH PC, PC <- PC + sign_extend(rel8)
BRA rel8 ; PC <- PC + sign_extend(rel8)

BSR rel16 ; PUSH PC, PC <- PC + rel16
BRA rel16 ; PC <- PC + rel16

BNE rel8 ; IF Z=0 THEN PC <- PC + sign_extend(rel8)
BCC rel8 ; IF C=0 THEN PC <- PC + sign_extend(rel8)
BVC rel8 ; IF V=0 THEN PC <- PC + sign_extend(rel8)
BPL rel8 ; IF N=0 THEN PC <- PC + sign_extend(rel8)
BEQ rel8 ; IF Z=1 THEN PC <- PC + sign_extend(rel8)
BCS rel8 ; IF C=1 THEN PC <- PC + sign_extend(rel8)
BVS rel8 ; IF V=1 THEN PC <- PC + sign_extend(rel8)
BMI rel8 ; IF N=1 THEN PC <- PC + sign_extend(rel8)

BNE rel16 ; IF Z=0 THEN PC <- PC + rel16
BCC rel16 ; IF C=0 THEN PC <- PC + rel16
BVC rel16 ; IF V=0 THEN PC <- PC + rel16
BPL rel16 ; IF N=0 THEN PC <- PC + rel16
BEQ rel16 ; IF Z=1 THEN PC <- PC + rel16
BCS rel16 ; IF C=1 THEN PC <- PC + rel16
BVS rel16 ; IF V=1 THEN PC <- PC + rel16
BMI rel16 ; IF N=1 THEN PC <- PC + rel16
